── 2. Inside #region Invoke Helpers ───────────────────────────────────────────────

    public static void RaiseGameplayEvent(GameplayEvent gameplayEvent) =>
        OnGameplayEvent?.Invoke(gameplayEvent);

    public static void QuestAccepted(ControllerBrain brain, string questId) =>
        OnQuestAccepted?.Invoke(brain, questId);

    public static void QuestProgress(ControllerBrain brain, string questId) =>
        OnQuestProgress?.Invoke(brain, questId);

    public static void QuestStageChanged(ControllerBrain brain, string questId, int stageIndex) =>
        OnQuestStageChanged?.Invoke(brain, questId, stageIndex);

    public static void QuestCompleted(ControllerBrain brain, string questId) =>
        OnQuestCompleted?.Invoke(brain, questId);

    public static void QuestFailed(ControllerBrain brain, string questId) =>
        OnQuestFailed?.Invoke(brain, questId);

    public static void QuestAbandoned(ControllerBrain brain, string questId) =>
        OnQuestAbandoned?.Invoke(brain, questId);
